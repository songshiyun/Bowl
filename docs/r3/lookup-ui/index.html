FUSE fuse_entry_param 与 LookupResponse 结构映射

FUSE 内核参数 (fuse_entry_param) Stronger Proto 映射字段 类型 设计说明与作用
ino (fuse_ino_t) inode.ino uint64 目标实体的全局唯一索引节点标识。根目录通常为 1。
generation (uint64) inode.generation uint64 防止 Inode 回收复用导致脏缓存击穿的代数版本。
attr.st_mode (mode_t) inode.attr.mode uint32 文件类型掩码 (S_IFDIR, S_IFREG) 与 POSIX 权限位 (0755, 0644)。
attr.st_size (off_t) inode.attr.size uint64 文件物理大小（字节数）。
attr.st_nlink (nlink_t) inode.attr.nlink uint32 硬链接数。普通文件默认 1；目录至少为 2（含 . 与 ..）。
attr.st_uid / st_gid inode.attr.uid / gid uint32 文件所属的用户 ID 与用户组 ID。
attr.st_atim / mtim / ctim inode.attr.atime / mtime / ctime google.protobuf.Timestamp 访问时间、修改时间与状态变更时间（纳秒级）。
entry_timeout (double) entry_timeout_seconds double 告诉 Linux 内核可以在 VFS dentry 缓存该名称映射多少秒。
attr_timeout (double) attr_timeout_seconds double 告诉 Linux 内核可以在 VFS 缓存该 Inode 属性多少秒。

syntax = "proto3";

package stronger.v1;

option go_package = "github.com/starsrc/bowl/gen/go/stronger/v1;strongerv1";

import "buf/validate/validate.proto";
import "google/protobuf/timestamp.proto";

// StrongerService 提供针对分布式文件系统的元数据操作与 FUSE 协议对接能力。
service StrongerService {
  // Lookup 执行单级目录项查找。
  // 根据父目录 Inode 与目标子项名称获取对应的 Inode 元数据及内核缓存时效。
  // 若目标子项不存在，必须返回 gRPC 标准状态码 NotFound (映射至 POSIX ENOENT)。
  rpc Lookup (LookupRequest) returns (LookupResponse);
}

// LookupRequest 定义 Lookup 接口输入参数。
message LookupRequest {
  // 父目录的 Inode 编号。根目录通常为 1。
  uint64 parent_ino = 1 [(buf.validate.field).uint64.gt = 0];

  // 待查找的子项名称（文件或目录名）。
  // 长度必须在 1 到 255 字节之间，且严禁包含路径分隔符 '/'。
  string name = 2 [
    (buf.validate.field).string.min_len = 1,
    (buf.validate.field).string.max_bytes = 255,
    (buf.validate.field).string.not_contains = "/"
  ];
}

// LookupResponse 包含查找命中的节点元数据以及 FUSE 内核缓存有效时间。
message LookupResponse {
  // 命中的目标 Inode 实体元数据。
  Inode inode = 1 [(buf.validate.field).required = true];

  // 目录项 (dentry) 在 FUSE 内核中的有效缓存时长（秒）。
  double entry_timeout_seconds = 2 [(buf.validate.field).double.gte = 0];

  // 属性 (attr) 在 FUSE 内核中的有效缓存时长（秒）。
  double attr_timeout_seconds = 3 [(buf.validate.field).double.gte = 0];
}

// Inode 代表文件系统中的实体节点。
message Inode {
  // 节点的全局唯一持久化标识。
  uint64 ino = 1 [(buf.validate.field).uint64.gt = 0];

  // Inode 代数版本，用于标识 Inode 复用生命周期，防止陈旧引用。
  uint64 generation = 2;

  // 节点的 POSIX 物理属性集合。
  InodeAttribute attr = 3 [(buf.validate.field).required = true];
}

// InodeAttribute 包含遵循 POSIX 规范的节点物理属性。
message InodeAttribute {
  // 文件的模式位（包含文件类型 S_IFREG, S_IFDIR 及权限位 0755, 0644 等）。
  uint32 mode = 1 [(buf.validate.field).uint32.gt = 0];

  // 文件物理大小（字节数）。
  uint64 size = 2;

  // 硬链接数量。文件默认为 1；目录通常至少为 2。
  uint32 nlink = 3 [(buf.validate.field).uint32.gt = 0];

  // 所属用户 ID。
  uint32 uid = 4;

  // 所属组 ID。
  uint32 gid = 5;

  // 特殊设备号（仅当为设备文件时有效，常规文件为 0）。
  uint32 rdev = 6;

  // 块大小（字节数，例如 4096）。
  uint32 blksize = 7;

  // 占用的 512 字节块数量。
  uint64 blocks = 8;

  // 最后访问时间。
  google.protobuf.Timestamp atime = 9;

  // 最后修改时间。
  google.protobuf.Timestamp mtime = 10;

  // 状态变更时间。
  google.protobuf.Timestamp ctime = 11;
}

📖 领域术语表 (Domain Glossary)

遵循 ASD-STE100 规范与 Domain Modeling 原则

Inode (索引节点)

文件系统中标识文件或目录实体的唯一持久化数字标识符与元数据载体。

_Avoid_: FileID, NodeID, ObjectId, VNode

Entry (目录项)

父目录下将文件名映射至目标 Inode 的单向命名绑定关系。

_Avoid_: Dentry, FileItem, DirRecord, Record

Lookup (路径查找)

根据父级 Inode 与目标名称，解析并获取对应 Entry 及其 Inode 属性的元数据检索操作。

_Avoid_: Find, Search, GetFile, Query

Attribute (文件属性)

描述 Inode 物理特征与内核访问控制参数的集合（如大小、权限掩码、硬链接数、时间戳等）。

_Avoid_: Stat, Meta, Properties, Info

Generation (代数)

用于标识 Inode 复用生命周期的递增版本号，防止客户端缓存引用已释放后重新分配的 Inode。

_Avoid_: Version, Revision, Epoch
